import json, copy
x = json.load(open('TASKS.json'))
tasks = {t['id']: t for t in x['tasks']}
assert len(tasks) == len(x['tasks'])
def closure(tasks):
    cache, visiting = {}, set()
    def visit(id):
        assert id not in visiting, ('cycle', id)
        if id in cache: return cache[id]
        visiting.add(id)
        out = set()
        for dep in tasks[id]['deps']:
            assert dep in tasks, ('missing', dep)
            out |= {dep} | visit(dep)
        visiting.remove(id)
        cache[id] = out
        return out
    for id in tasks: visit(id)
    return cache
required = {

 'T0707': {'T0502','T0504','T0508','T0709','T0805','T0806','T0807'},
